cmake_minimum_required(VERSION 3.28)
set(PICO_BOARD waveshare_rp2350_one CACHE STRING "Target board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "Target architecture")
include(pico_sdk_import.cmake)
project(pico_all VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_executable(pico_all)
set(USB_VID 0xFEFF CACHE STRING "USB vendor ID for local development")
set(USB_PID 0xFCFD CACHE STRING "USB product ID for local development")
set(USB_ITF_HID 1)
set(USB_ITF_CCID 1)
set(USB_ITF_WCID 1)
set(ENABLE_LIBCVC 1)
set(ENABLE_EDDSA ON CACHE BOOL "Enable Ed25519 and Ed448")
set(ENABLE_PQC OFF CACHE BOOL "Enable post-quantum algorithms")
set(ENABLE_POWER_ON_RESET ON)
add_compile_definitions(ENABLE_OATH_APP=1 ENABLE_OTP_APP=1 ENABLE_POWER_ON_RESET=1)
include(pico-keys-sdk/picokeys_sdk_import.cmake)
# picokeys_sdk carries its sources via INTERFACE; do not compile them twice.
target_link_libraries(pico_all PRIVATE picokeys_sdk)
target_sources(pico_all PRIVATE src/integration.c)
target_include_directories(pico_all PRIVATE ${INCLUDES})
foreach(component IN ITEMS fido openpgp hsm)
    file(GLOB component_sources CONFIGURE_DEPENDS src/${component}/*.c)
    list(FILTER component_sources EXCLUDE REGEX "/defs\\.c$")
    if(component STREQUAL "openpgp")
        list(FILTER component_sources EXCLUDE REGEX "/management\\.c$")
        file(GLOB piv_sources CONFIGURE_DEPENDS src/piv/*.c)
        list(APPEND component_sources ${piv_sources})
    endif()
    target_sources(pico_all PRIVATE ${component_sources})
    set_source_files_properties(${component_sources} PROPERTIES
        INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/src/${component};${CMAKE_CURRENT_SOURCE_DIR}/src/piv"
        COMPILE_OPTIONS "-Wall;-Wextra"
    )
endforeach()
# Use OpenPGP's standard PIN semantics by default; optional upstream admin-less mode is disabled.
pico_set_program_name(pico_all "Pico All")
pico_set_program_version(pico_all "0.1.0")
pico_set_binary_version(pico_all MAJOR 0 MINOR 1)
pico_enable_stdio_usb(pico_all 0)
pico_enable_stdio_uart(pico_all 0)
pico_add_extra_outputs(pico_all)
